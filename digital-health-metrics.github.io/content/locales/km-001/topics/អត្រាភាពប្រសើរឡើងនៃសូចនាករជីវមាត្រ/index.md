# អត្រាភាពប្រសើរឡើងនៃសូចនាករជីវមាត្រ

អត្រាភាពប្រសើរឡើងនៃសូចនាករជីវមាត្រ (biometric improvement rate) គឺជាសមាមាត្រនៃអ្នកជំងឺដែលបានចុះឈ្មោះក្នុងកម្មវិធីសុខភាពឌីជីថល ដែលសម្រេចបានភាពប្រសើរឡើងមានអត្ថន័យគ្លីនិកក្នុងសូចនាករជីវមាត្រដែលតាមដាន — ជាទូទៅអេម៉ូក្លូប៊ីនគ្លីកូស៊ីល (HbA1c) ក្នុងកម្មវិធីទឹកនោមផ្អែម និងកម្មវិធីបេះដូង-មេតាប៉ូលីក ឬសន្ទស្សន៍ម៉ាសរាងកាយ (BMI) ក្នុងកម្មវិធីគ្រប់គ្រងទម្ងន់ — ក្នុងរយៈពេលចុះឈ្មោះដែលបានកំណត់។ វាជារង្វាស់លទ្ធផលដែលបញ្ជាក់ទីបំផុតនូវការអះអាងគ្លីនិករបស់ផលិតផលសុខភាពឌីជីថល៖ តួលេខការចូលរួម និងការទទួលយកពិពណ៌នាអំពីរបៀបប្រើផលិតផល ប៉ុន្តែភាពប្រសើរឡើងនៃសូចនាករជីវមាត្រជិតនឹងភស្តុតាងថាវាដំណើរការ។

## ហេតុអ្វីបានជាសំខាន់

កម្មវិធីសុខភាពឌីជីថលតែងត្រូវបានលក់ និងបញ្ជាទិញដោយផ្អែកលើការសន្យាអំពីលទ្ធផលសុខភាពប្រសើរឡើង ហើយអត្រាភាពប្រសើរឡើងនៃសូចនាករជីវមាត្រគឺជាវិធីផ្ទាល់ និងអាចវាស់បរិមាណបានបំផុតដើម្បីសាកល្បងការសន្យានោះធៀបនឹងកម្រិតជាក់លាក់ដែលគ្លីនិកទទួលស្គាល់ ជាជាងការអះអាងស្រពិចស្រពិលអំពី "សុខភាពប្រសើរជាង"។ អ្នកទូទាត់ និយោជក និងប្រព័ន្ធសុខភាពកាន់តែភ្ជាប់សំណង ឬការបន្តកិច្ចសន្យាទៅនឹងការផ្លាស់ប្តូរសូចនាករជីវមាត្រដែលបានបង្ហាញ ដូច្នេះកម្មវិធីដែលមិនអាចរាយការណ៍អត្រានេះបានគួរឱ្យទុកចិត្តស្ថិតក្នុងភាពអន់ថយផ្នែកពាណិជ្ជកម្ម ក៏ដូចជាគ្លីនិក។ រង្វាស់នេះក៏ជាការត្រួតពិនិត្យវិន័យលើការរចនាកម្មវិធីផងដែរ៖ វាងាយរាយការណ៍ការចូលរួម (ការចូលប្រើ សារដែលបានផ្ញើ) ជាងលទ្ធផល ហើយក្រុមគួរសង្ស័យកម្មវិធីណាដែលរាយការណ៍ពីអតីតយ៉ាងក្លៀវក្លា ខណៈមិនច្បាស់លាស់អំពីក្រោយ។

## ឧទាហរណ៍អនុវត្ត

កម្មវិធីសុខភាពឌីជីថលបេះដូង-មេតាប៉ូលីកមួយចុះឈ្មោះអ្នកជំងឺ ៨០០ នាក់ដែលមាន HbA1c មូលដ្ឋានក្រៅចន្លោះ។ ក្នុងចំណោមនោះ ៦២០ នាក់មានទាំងការវាស់មូលដ្ឋានត្រឹមត្រូវ និងការវាស់តាមដាននៅ ៦ ខែ (១៨០ នាក់បាត់បង់ពីការតាមដាន ហើយត្រូវដកចេញពីភាគបែង មិនរាប់ជាបរាជ័យ)។ ក្នុងចំណោម ៦២០ នាក់ដែលមានការវាស់ជាគូ ៣៤០ នាក់សម្រេចបានការថយចុះយ៉ាងហោចណាស់ 0.5 ពិន្ទុភាគរយ។ អត្រាភាពប្រសើរឡើងនៃសូចនាករជីវមាត្រគឺ 340 / 620 × 100 = 55%។ ការរាយការណ៍ធៀបនឹងអ្នកចុះឈ្មោះទាំង ៨០០ (340 / 800 = 42.5%) នឹងច្រឡំការបាត់បង់ពីការតាមដានជាមួយការបរាជ័យនៃការព្យាបាល ដែលបង្ហាញអត្រាតិចជាងការពិតសម្រាប់អ្នកជំងឺដែលបានបញ្ចប់ការវាស់ពិតប្រាកដ។

## ប្រភពទិន្នន័យ និងចំណុចត្រូវប្រុងប្រយ័ត្ន

តម្លៃសូចនាករជីវមាត្រមូលដ្ឋាន និងតាមដានជាធម្មតាមកពីឧបករណ៍ភ្ជាប់ (ឧបករណ៍វាស់ជាតិស្ករ Bluetooth ឬជញ្ជីងឆ្លាតវៃ) លទ្ធផលមន្ទីរពិសោធន៍ដែលនាំចូលពីកំណត់ត្រាសុខភាពអេឡិចត្រូនិក ឬតម្លៃដែលអ្នកជំងឺរាយការណ៍ដោយខ្លួនឯង ហើយប្រភពទាំងបីនេះមានភាពអាចទុកចិត្តខុសគ្នាខ្លាំង ដូច្នេះគួររាយការណ៍ប្រភពរួមជាមួយអត្រា។ ការបាត់បង់ពីការតាមដានកម្រចៃដន្យ៖ អ្នកជំងឺដែលឈប់ចូលរួមកម្មវិធីតែងជាអ្នកដែលតិចនឹងប្រសើរឡើងបំផុត ដូច្នេះអត្រាភាពប្រសើរឡើងខ្ពស់ដែលគណនាលើតែអ្នកជំងឺដែលបានបញ្ចប់ការតាមដាន អាចបំផ្លើសឥទ្ធិពលកម្រិតប្រជាជនពិតរបស់កម្មវិធី។ ឥទ្ធិពលតាមរដូវ និងការត្រឡប់ទៅរកមធ្យម (regression to the mean) មានពិតសម្រាប់ទាំង HbA1c និងទម្ងន់ ដូច្នេះកម្មវិធីគួរប្រៀបធៀបជាមួយក្រុមត្រួតពិនិត្យដំណាលគ្នា ឬប្រវត្តិសាស្ត្រនៅកន្លែងអាចធ្វើបាន ជាជាងចាត់ទុកភាពប្រសើរឡើងណាមួយជាភស្តុតាងនៃឥទ្ធិពលរបស់កម្មវិធី។

## កំហុសដែលតែងកើតមាន

- **ដកចេញ ជាជាងរាយការណ៍ការបាត់បង់ពីការតាមដាន**៖ ការដកអ្នកជំងឺដែលគ្មានការវាស់តាមដានចេញពីភាគបែងដោយស្ងៀមស្ងាត់ អាចបំប៉នអត្រាភាពប្រសើរឡើងដែលមើលឃើញបានយ៉ាងខ្លាំង ដូច្នេះគួររាយការណ៍អត្រាបញ្ចប់ការវាស់តាមដានរួមជាមួយអត្រាភាពប្រសើរឡើងខ្លួនឯង។
- **លាយការវាស់ដែលរាយការណ៍ដោយខ្លួនឯង និងមកពីឧបករណ៍ ដោយមិនដាក់ស្លាក**៖ ទម្ងន់ដែលរាយការណ៍ដោយខ្លួនឯងមិនសូវអាចទុកចិត្តបានជាប្រព័ន្ធ ជាងការអានពីជញ្ជីងឆ្លាតវៃដែលភ្ជាប់ ហើយការលាយប្រភពទាំងពីរបិទបាំងថាភាពប្រសើរឡើងដែលមើលឃើញប៉ុនណាជាសំឡេងរំខាននៃការវាស់។
- **គ្មានក្រុមត្រួតពិនិត្យ ឬការប្រៀបធៀបផ្ទុយ**៖ រង្វាស់សូចនាករជីវមាត្ររ៉ាំរ៉ៃជាច្រើនប្រែប្រួល ឬត្រឡប់ទៅរកមធ្យមដោយខ្លួនឯង ដូច្នេះអត្រាភាពប្រសើរឡើងក្រុមតែមួយដោយគ្មានក្រុមប្រៀបធៀបគឺជាភស្តុតាងចង្អុលបង្ហាញ មិនមែនសម្រេចនៃឥទ្ធិពលកម្មវិធីទេ។
- **ចាត់ទុកការផ្លាស់ប្តូរមធ្យមតិចតួចជាភស្តុតាងនៃភាពប្រសើរឡើងទូលំទូលាយ**៖ ភាពប្រសើរឡើងមធ្យមកម្រិតប្រជាជនតិចតួចអាចត្រូវបានជំរុញដោយអ្នកឆ្លើយតបខ្លាំងមួយចំនួន ខណៈអ្នកជំងឺភាគច្រើនមិនឃើញការផ្លាស់ប្តូរ ដូច្នេះគួររាយការណ៍ការចែកចាយ (ឧ. សមាមាត្រដែលឆ្លងកម្រិតមានអត្ថន័យគ្លីនិក) មិនត្រឹមការផ្លាស់ប្តូរមធ្យម។
