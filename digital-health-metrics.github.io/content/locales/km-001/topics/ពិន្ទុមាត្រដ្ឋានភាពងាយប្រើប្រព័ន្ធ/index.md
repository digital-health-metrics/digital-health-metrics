# ពិន្ទុមាត្រដ្ឋានភាពងាយប្រើប្រព័ន្ធ (System Usability Scale)

ពិន្ទុមាត្រដ្ឋានភាពងាយប្រើប្រព័ន្ធ (System Usability Scale ឬ SUS) គឺជាកម្រងសំណួរស្តង់ដារ ១០ ចំណុច ដែលប្រើដើម្បីវាស់បរិមាណថាកម្មវិធីមួយងាយប្រើប្រាស់កម្រិតណា ដោយផលិតពិន្ទុតែមួយពី 0 ដល់ 100 ដែលអាចប្រៀបធៀបនឹងបទដ្ឋានឧស្សាហកម្មដែលបានបង្កើតយ៉ាងរឹងមាំ។ ខុសពី Net Promoter Score ដែលវាស់ឆន្ទៈក្នុងការណែនាំបន្ត ឬរង្វាស់លទ្ធផលដែលរាយការណ៍ដោយអ្នកជំងឺ ដែលវាស់ស្ថានភាពគ្លីនិក ឬមុខងារ SUS វាស់រឿងជាក់លាក់មួយ៖ កម្មវិធីខ្លួនឯងងាយរៀន និងងាយប្រើប្រាស់កម្រិតណា សម្រាប់អ្នកជំងឺ ឬបុគ្គលិកគ្លីនិក។

## ហេតុអ្វីបានជាសំខាន់

ឧបករណ៍សុខភាពឌីជីថលអាចមានភស្តុតាងគ្លីនិករឹងមាំ និងករណីអាជីវកម្មដ៏គួរឱ្យជឿជាក់ ខណៈនៅតែបរាជ័យក្នុងការអនុវត្តជាក់ស្តែង ដោយសារអ្នកជំងឺ ឬគ្លីនិកករយល់ថាចំណុចប្រទាក់សាមញ្ញច្របូកច្របល់ យឺត ឬគួរឱ្យខកចិត្តក្នុងការប្រើ — ហើយដោយសារ SUS ជាឧបករណ៍ដែលមានសុពលភាព និងប្រើទូលំទូលាយ ជាមួយទិន្នន័យប្រៀបធៀបដែលបានបោះពុម្ពជាច្រើនទសវត្សរ៍ឆ្លងឧស្សាហកម្ម វាអនុញ្ញាតឱ្យក្រុមសុខភាពឌីជីថលប្រៀបធៀបភាពងាយប្រើនៃផលិតផលរបស់ខ្លួនធៀបនឹងការចែកចាយដែលស្គាល់ ជាជាងពឹងលើចំណាប់អារម្មណ៍មិនផ្លូវការ ឬការត្អូញត្អែរតាមអ្នកដទៃប្រាប់។ SUS មិនអាស្រ័យលើបច្ចេកវិទ្យាដោយចេតនា និងលឿនក្នុងការអនុវត្ត (ជាធម្មតាក្រោមប្រាំនាទី) ដែលធ្វើឱ្យមានភាពជាក់ស្តែងក្នុងការអនុវត្តម្តងហើយម្តងទៀតឆ្លងការធ្វើឡើងវិញនៃការរចនា ខុសពីការសិក្សាភាពងាយប្រើពេញលេញ ឬការសាកល្បងគ្លីនិកផ្លូវការ។ ដោយសារការបរាជ័យនៃភាពងាយប្រើសម្រាប់គ្លីនិកករគឺជាកត្តារួមចំណែកដែលមានឯកសារបញ្ជាក់ដល់ការអស់កម្លាំងការងារ (សូមមើលអត្រាអស់កម្លាំងការងាររបស់វេជ្ជបណ្ឌិត) ហើយការបរាជ័យនៃភាពងាយប្រើសម្រាប់អ្នកជំងឺគឺជាកត្តារួមចំណែកដែលមានឯកសារបញ្ជាក់ដល់ការបោះបង់ចោល និងលទ្ធផលចំណេះដឹងឌីជីថលខ្សោយ (សូមមើលអត្រាចំណេះដឹងឌីជីថល) SUS ដើរតួជាសញ្ញាព្រមានដំបូងថ្លៃទាបនៃភាពងាយប្រើ ដែលអាចចាប់បញ្ហារចនាបានមុនពេលវាលេចឡើងក្នុងរង្វាស់ខាងក្រោមដែលមានផលវិបាកធំជាងនោះ។

## របៀបគណនា

```

              (25 − ផលបូកពិន្ទុធាតុលេខគូ)) × 2.5

លទ្ធផលជាពិន្ទុតែមួយពី 0 ដល់ 100 (មិនមែនភាគរយទេ ទោះបីមាត្រដ្ឋានដូច្នេះ
ដោយសារវាមិនតំណាង "ភាគរយត្រឹមត្រូវ" ឬស្រដៀងគ្នា)។

  ក្រោម 51   — ភាពងាយប្រើខ្សោយ ត្រូវការស៊ើបអង្កេត
```

## ឧទាហរណ៍អនុវត្ត

វេទិកាសុខភាពពីចម្ងាយមួយចែកកម្រងសំណួរ SUS ស្តង់ដារ ១០ ចំណុចទៅអ្នកជំងឺ ១៥០ នាក់ក្រោយការជួបតាមវីដេអូលើកដំបូងរបស់ពួកគេ។ ពិន្ទុ SUS មធ្យមដែលបានគណនាពីអ្នកឆ្លើយទាំងអស់គឺ 74។ ធៀបនឹងមធ្យមឧស្សាហកម្មដែលបានដកស្រង់ទូលាយ 68 នេះបង្ហាញភាពងាយប្រើខ្ពស់ជាងមធ្យមសម្រាប់ប្រជាជនអ្នកជំងឺ និងករណីប្រើជាក់លាក់នេះ ទោះបីនៅតែទាបជាងកម្រិត "ល្អឥតខ្ចោះ" 80 ដែលនឹងបង្ហាញថានៅសល់ឧបសគ្គភាពងាយប្រើតិចតួចក៏ដោយ។ ការបែងចែកការឆ្លើយតប ១៥០ ដដែលតាមអាយុបង្ហាញពិន្ទុមធ្យម 81 សម្រាប់អ្នកជំងឺអាយុក្រោម ៥០ និង 62 សម្រាប់អ្នកជំងឺអាយុ ៦៥ ឡើងទៅ — គម្លាតដែលចង្អុលទៅបញ្ហាភាពងាយប្រើជាក់លាក់ដែលអាចដោះស្រាយបានសម្រាប់អ្នកជំងឺវ័យចាស់ ជាជាងបញ្ហាភាពងាយប្រើទូទៅនៃផលិតផល ហើយជាបញ្ហាដែលមធ្យមលាយតែមួយនឹងលាក់។

## ប្រភពទិន្នន័យ និងចំណុចត្រូវប្រុងប្រយ័ត្ន

ទិន្នន័យ SUS មកដោយផ្ទាល់ពីអ្នកជំងឺ ឬគ្លីនិកករដែលបំពេញកម្រងសំណួរស្តង់ដារ ១០ ចំណុច ហើយឧបករណ៍ត្រូវតែអនុវត្តយ៉ាងពិតប្រាកដដូចសុពលភាព (ធាតុ ១០ ដូចគ្នា មាត្រដ្ឋានយល់ព្រម ៥ ចំណុចដូចគ្នា រូបមន្តដាក់ពិន្ទុដូចគ្នា) ដើម្បីឱ្យពិន្ទុលទ្ធផលអាចប្រៀបធៀបនឹងមូលដ្ឋានដែលបានបោះពុម្ពបាន។ កំណែដែលកែប្រែ ឬខ្លីជាងនៃកម្រងសំណួរ ទោះមានបំណងល្អប៉ុនណា បង្កើតពិន្ទុដែលមិនអាចបកស្រាយបានដោយភាពអាចទុកចិត្តធៀបនឹងការចែកចាយមូលដ្ឋានស្តង់ដារ។ SUS វាស់ភាពងាយប្រើដែលបានយល់ឃើញ ដែលទាក់ទងនឹងប៉ុន្តែមិនដូចគ្នានឹងភាពជោគជ័យនៃការបំពេញកិច្ចការដែលគ្មានលំអៀង (សូមមើលអត្រាចំណេះដឹងឌីជីថលសម្រាប់រង្វាស់ផ្អែកលើការបំពេញកិច្ចការ)៖ ផលិតផលអាចមានពិន្ទុ SUS ល្អពីអ្នកជំងឺដែលមិនបានព្យាយាមមុខងារស្មុគស្មាញជាង ដូច្នេះការផ្គូផ្គង SUS ជាមួយទិន្នន័យបំពេញកិច្ចការគ្មានលំអៀង ផ្តល់រូបភាពពេញលេញជាងមួយណាមួយតែឯង។ ពេលវេលាឆ្លើយតបមានសារៈសំខាន់៖ ការចែក SUS ភ្លាមៗក្រោយឧប្បត្តិហេតុជាក់លាក់គួរឱ្យខកចិត្ត (ការភ្ជាប់បរាជ័យ ជំហានមិនច្បាស់) ធៀបនឹងក្រោយវគ្គរលូន អាចផ្លាស់ប្តូរពិន្ទុដោយឯករាជ្យពីភាពងាយប្រើរួមរបស់ផលិតផល។

## កំហុសដែលតែងកើតមាន

- **កែប្រែធាតុកម្រងសំណួរស្តង់ដារ ឬការដាក់ពិន្ទុ**៖ សូម្បីតែការផ្លាស់ប្តូរពាក្យ ឬមាត្រដ្ឋានតិចតួចក៏ធ្វើឱ្យការប្រៀបធៀបនឹងការចែកចាយមូលដ្ឋានដែលបានបោះពុម្ពរឹងមាំមិនមានសុពលភាព ដូច្នេះគួរប្រើឧបករណ៍ស្តង់ដារ ១០ ធាតុ យ៉ាងពិតប្រាកដតាមសុពលភាព។
- **រាយការណ៍តែពិន្ទុមធ្យម ដោយមិនបែងចែក**៖ ភាពងាយប្រើតែងប្រែប្រួលច្រើនតាមអាយុអ្នកប្រើ ចំណេះដឹងឌីជីថល ឬតួនាទី (អ្នកជំងឺ ធៀបនឹងគ្លីនិកករ) ដូច្នេះគួររាយការណ៍បែងចែក ដើម្បីរកគម្លាតភាពងាយប្រើជាក់លាក់ដែលអាចដោះស្រាយបាន ដែលមធ្យមតែមួយលាក់។
- **ចាត់ទុក SUS ជារង្វាស់ប្រសិទ្ធភាពគ្លីនិក**៖ SUS វាស់ភាពងាយប្រើជាពិសេស មិនមែនលទ្ធផលគ្លីនិក ឬភាពពេញចិត្តនឹងការថែទាំទេ ឧបករណ៍ងាយប្រើខ្លាំងនៅតែអាចបរាជ័យក្នុងការកែលម្អលទ្ធផលគ្លីនិក ហើយទាំងពីរមិនគួរត្រូវបានច្រឡំ ឬជំនួសគ្នាឡើយ។
- **ចែកការស្ទង់មតិតែក្រោយវគ្គរលូនមិនធម្មតា ឬគួរឱ្យខកចិត្តមិនធម្មតា**៖ ពេលវេលា និងបរិបទនៃការចែកអាចធ្វើឱ្យពិន្ទុលំអៀង ដូច្នេះគួរចែកជាប់លាប់ឆ្លងគំរូតំណាងនៃវគ្គក្នុងពិភពពិត មិនមែនតែវគ្គងាយស្រួល ឬវគ្គជ្រើសរើសពិសេសប៉ុណ្ណោះ។
