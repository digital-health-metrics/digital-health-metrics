# អត្រាភាពទៀងទាត់នៃការចូលរួមរបស់អ្នកជំងឺ

អត្រាភាពទៀងទាត់នៃការចូលរួមរបស់អ្នកជំងឺ (patient engagement consistency rate) វាស់ថាអ្នកជំងឺដែលបានចុះឈ្មោះធ្វើអន្តរកម្មជាមួយផលិតផលសុខភាពឌីជីថលទៀងទាត់កម្រិតណាតាមពេលវេលា ឧទាហរណ៍ការកត់ត្រាអាហារ ឬរោគសញ្ញា ការកត់ត្រាសកម្មភាពរាងកាយ ឬការមើលទិន្នន័យសុខភាព ជាជាងគ្រាន់តែថាតើពួកគេធ្លាប់ប្រើវាឬអត់។ វាជារង្វាស់រយៈពេលវែង ខុសពីចំនួនអ្នកប្រើសកម្មនៅពេលណាមួយ៖ អ្នកជំងឺពីរនាក់អាចមានស្ថានភាព "ប្រើកម្មវិធីខែនេះ" ដូចគ្នា ខណៈដែលម្នាក់កត់ត្រាជាប់លាប់រាល់ថ្ងៃ ហើយម្នាក់ទៀតកត់ត្រាតែម្តងរួចបាត់ខ្លួនបីសប្តាហ៍ ហើយមានតែរង្វាស់ភាពទៀងទាត់ប៉ុណ្ណោះដែលញែកពួកគេចេញពីគ្នា។

## ហេតុអ្វីបានជាសំខាន់

អន្តរកម្មទៀងទាត់ជាប់លាប់ជាមួយឧបករណ៍សុខភាពឌីជីថល គឺជាសូចនាករនាំមុខដែលអាចទុកចិត្តបានបំផុតមួយនៃអត្ថប្រយោជន៍គ្លីនិក ជាពិសេសសម្រាប់ជំងឺដែលអាស្រ័យលើឥរិយាបថ ដូចជាទឹកនោមផ្អែម ការគ្រប់គ្រងទម្ងន់ និងសុខភាពផ្លូវចិត្ត ដែលតម្លៃរបស់ឧបករណ៍កើតពីទម្លាប់ដែលវាគាំទ្រ ជាជាងវគ្គតែមួយ។ ផលិតផលអាចរាយការណ៍ចំនួនអ្នកប្រើសកម្មប្រចាំខែដែលមើលទៅល្អ ខណៈពិតជាបម្រើប្រជាជនដែលចូលម្តងរួចរសាត់ទៅ ព្រោះការប្រើសកម្មប្រចាំខែជាបន្ទាត់ទាបដែលមិនប្រាប់អ្វីអំពីលំនាំប្រើប្រាស់ក្នុងខែ។ រង្វាស់ភាពទៀងទាត់ចាប់បានរឿងនេះតាមរបៀបដែលចំនួនសកម្មភាពសាមញ្ញមិនអាចធ្វើបាន។ ដោយសារភាពទៀងទាត់ក៏ជារឿងលំបាកជាងក្នុងការរក្សាអស់ជាច្រើនខែ ជាជាងសប្តាហ៍ វាជាសញ្ញាស្មោះត្រង់ជាងអំពីគុណភាពផលិតផល និងភាពសមស្របផ្នែកគ្លីនិក ជាងតួលេខការចូលរួមរយៈពេលខ្លី ដែលងាយទទួលឥទ្ធិពលនៃភាពថ្មីថ្មោងភ្លាមៗបន្ទាប់ពីការចូលរួមដំបូង។

## របៀបគណនា

```
អត្រាភាពទៀងទាត់នៃការចូលរួម = សប្តាហ៍ដែលមានអន្តរកម្មមានលក្ខណៈសម្បត្តិយ៉ាងហោចណាស់មួយ
                               / សប្តាហ៍សរុបដែលបានចុះឈ្មោះ × 100

"អន្តរកម្មមានលក្ខណៈសម្បត្តិ" គួរត្រូវបានកំណត់ឱ្យច្បាស់ និងជាប់លាប់
(ឧ. ការកត់ត្រាអាហារ ការពិនិត្យរោគសញ្ញា ឬការធ្វើសមកាលកម្មសកម្មភាពដែលបានបញ្ចប់) —
មិនមែនព្រឹត្តិការណ៍អកម្មដូចជាការបើកកម្មវិធីដោយគ្មានសកម្មភាពកត់ត្រាទេ។

រាយការណ៍ជាការចែកចាយ មិនមែនត្រឹមមធ្យមប្រជាជនប៉ុណ្ណោះ៖
  ឧ. សមាមាត្រអ្នកជំងឺដែលមានភាពទៀងទាត់ប្រចាំសប្តាហ៍ ≥ 80%
      សមាមាត្រដែលមាន 50-79% សមាមាត្រដែលមាន < 50%
```

## ឧទាហរណ៍អនុវត្ត

កម្មវិធីណែនាំអាហារូបត្ថម្ភមួយចុះឈ្មោះអ្នកជំងឺម្នាក់រយៈពេល ១២ សប្តាហ៍។ អ្នកជំងឺកត់ត្រាអាហារមានលក្ខណៈសម្បត្តិយ៉ាងហោចណាស់មួយក្នុង ៩ នៃ ១២ សប្តាហ៍នោះ ដែលផ្តល់អត្រាភាពទៀងទាត់នៃការចូលរួមបុគ្គល 9 / 12 × 100 = 75%។ ក្នុងក្រុមអ្នកជំងឺទាំងមូលចំនួន ២,០០០ នាក់របស់កម្មវិធីដែលបានចុះឈ្មោះយ៉ាងហោចណាស់ ១២ សប្តាហ៍ អ្នកជំងឺ ៦០០ នាក់ (30%) រក្សាភាពទៀងទាត់ប្រចាំសប្តាហ៍ ≥ 80% ៩០០ នាក់ (45%) ស្ថិតក្នុងកម្រិត 50-79% និង ៥០០ នាក់ (25%) ស្ថិតក្រោម 50%។ ការរាយការណ៍តែមធ្យមក្រុម (ដែលអាចចុះនៅប្រហែល 65%) នឹងបិទបាំងការពិតដែលថាមួយភាគបួនពេញនៃអ្នកជំងឺស្ទើរតែមិនចូលរួមសោះ ដែលជាផ្នែកមួយគួរស៊ើបអង្កេតដោយឡែក ជាជាងរំលាយចូលក្នុងមធ្យមរួម។

## ប្រភពទិន្នន័យ និងចំណុចត្រូវប្រុងប្រយ័ត្ន

ទិន្នន័យភាពទៀងទាត់មកពីកំណត់ហេតុព្រឹត្តិការណ៍របស់ផលិតផលផ្ទាល់ (ការកត់ត្រាអាហារ ការធ្វើសមកាលកម្មសកម្មភាព ការពិនិត្យ) ហើយនិយមន័យនៃ "អន្តរកម្មមានលក្ខណៈសម្បត្តិ" មានឥទ្ធិពលធំធេងលើអត្រាលទ្ធផល ដោយនិយមន័យរលុង (ការបើកកម្មវិធីណាមួយ) នឹងមើលទៅល្អជាងនិយមន័យតឹងរឹង (ការកត់ត្រាដែលបានបញ្ចប់ និងមានន័យ) ដូច្នេះនិយមន័យដែលប្រើត្រូវតែបញ្ជាក់ឱ្យច្បាស់ជាមួយតួលេខដែលរាយការណ៍។ ទិន្នន័យដែលធ្វើសមកាលកម្មដោយស្វ័យប្រវត្តិ (ឧទាហរណ៍ឧបករណ៍តាមដានសុខភាពដែលភ្ជាប់ធ្វើសមកាលកម្មសកម្មភាពនៅផ្ទៃខាងក្រោយ) គួរត្រូវបានរាយការណ៍ដោយឡែកពីទិន្នន័យដែលកត់ត្រាដោយដៃ ព្រោះការធ្វើសមកាលកម្មស្វ័យប្រវត្តិអាចបំប៉នភាពទៀងទាត់ដែលមើលឃើញ ដោយមិនឆ្លុះបញ្ចាំងពីការខិតខំប្រឹងប្រែងសកម្មរបស់អ្នកជំងឺ ឬការចូលរួមជាមួយការណែនាំរបស់ផលិតផលឡើយ។

## កំហុសដែលតែងកើតមាន

- **ច្រឡំការបើកកម្មវិធីជាមួយការចូលរួមមានន័យ**៖ ការបើកកម្មវិធីអកម្ម (ឧទាហរណ៍ត្រូវបានជំរុញដោយការជូនដំណឹង) មិនដូចការកត់ត្រាអាហារ ឬការពិនិត្យដែលបានបញ្ចប់ទេ ដូច្នេះគួរកំណត់ និងរាយការណ៍តែអន្តរកម្មមានលក្ខណៈសម្បត្តិប៉ុណ្ណោះ។
- **រាយការណ៍តែមធ្យមប្រជាជន**៖ អត្រាភាពទៀងទាត់មធ្យមដែលមើលទៅល្អអាចលាក់ប្រជាជនពីរភាគ គឺអ្នកចូលរួមខ្លាំង និងអ្នកស្ទើរតែមិនចូលរួមទាំងស្រុង ដូច្នេះគួររាយការណ៍ការចែកចាយតាមកម្រិតភាពទៀងទាត់ មិនត្រឹមតែមធ្យម។
- **មិនអើពើភាគបែងរយៈពេលចុះឈ្មោះ**៖ ការប្រៀបធៀបអត្រាភាពទៀងទាត់រវាងអ្នកជំងឺដែលបានចុះឈ្មោះរយៈពេលខុសគ្នាខ្លាំង ដោយមិនគិតពីរយៈពេលចុះឈ្មោះ នឹងលំអៀងទៅរកក្រុមណាដែលមានបង្អួចវាស់វែងខ្លីជាង និងរក្សាងាយជាង។
- **ការធ្វើសមកាលកម្មផ្ទៃខាងក្រោយស្វ័យប្រវត្តិបំប៉នអត្រា**៖ ស្ទ្រីមទិន្នន័យពាក់ក្នុងខ្លួនដែលធ្វើសមកាលកម្មអកម្ម អាចធ្វើឱ្យអ្នកជំងឺដែលឈប់ចូលរួមមើលទៅសកម្មជាប់លាប់ ដោយគ្មានការផ្លាស់ប្តូរឥរិយាបថពិត ឬការចូលរួមជាមួយផលិតផលរបស់ពួកគេ។
