# រយៈពេលដោះស្រាយការបញ្ជូនបន្តឌីជីថល

រយៈពេលដោះស្រាយការបញ្ជូនបន្តឌីជីថល (digital referral turnaround time) គឺជារយៈពេលដែលកន្លងផុតចាប់ពីការបញ្ជូនបន្តអេឡិចត្រូនិកត្រូវបានដាក់ស្នើដោយគ្លីនិកករបញ្ជូន រហូតដល់វាត្រូវបានចាត់ថ្នាក់អាទិភាព (triage) ហើយទទួលយក បដិសេធ ឬកក់ដោយសេវាទទួល។ វាជារង្វាស់ដំណើរការ (លំហូរ) ខុសពីរយៈពេលរង់ចាំសរុបរបស់អ្នកជំងឺ ហើយជាកន្លែងមួយក្នុងចំណោមកន្លែងច្បាស់បំផុតដែលការផ្លាស់ប្តូរប្រព័ន្ធឌីជីថល (ការបញ្ជូនបន្តអេឡិចត្រូនិកមានរចនាសម្ព័ន្ធ ការចាត់ថ្នាក់ដោយផ្អែកលើរូបភាព ទម្រង់បញ្ជូនបន្តស្តង់ដារ) អាចបង្ហាញថាបានផ្លាស់ប្តូរលេខប្រតិបត្តិការ ជាជាងគ្រាន់តែពិន្ទុភាពពេញចិត្ត។

## ហេតុអ្វីបានជាសំខាន់

ជំហានចាត់ថ្នាក់អាទិភាពយឺត ឬប្រែប្រួលខ្លាំង បន្ថែមការពន្យារពេលមុនពេលអ្នកជំងឺចូលបញ្ជីរង់ចាំគ្លីនិកផង ហើយដោយសារការពន្យារពេលនោះកើតឡើងមុនការថែទាំគ្លីនិកណាមួយចាប់ផ្តើម វាជាកាកសំណល់ដំណើរការសុទ្ធ ដែលឧបករណ៍ឌីជីថលស័ក្តិសមក្នុងការលុបចោល។ ប្រព័ន្ធបញ្ជូនបន្តដែលបង្ខំឱ្យមានវដ្ត "ត្រឡប់ទៅអ្នកបញ្ជូន" សម្រាប់ព័ត៌មានដែលខ្វះ បង្កើតរង្វិលកែតម្រូវដែលងាយមើលរំលង ប្រសិនបើរយៈពេលដោះស្រាយត្រូវបានវាស់តែលើការបញ្ជូនបន្តដែលឆ្លងកាត់ស្អាតបាតក្នុងលើកដំបូង។ នៅកន្លែងដែលសេវាបានណែនាំទម្រង់បញ្ជូនបន្តឌីជីថលមានរចនាសម្ព័ន្ធ វាលចាំបាច់ ឬការចាត់ថ្នាក់ផ្អែកលើរូបភាព (ឧទាហរណ៍ក្នុងសុខភាពស្បែកពីចម្ងាយ) រយៈពេលដោះស្រាយជាធម្មតាគឺជារង្វាស់តែមួយដែលមានអំណាចបញ្ចុះបញ្ចូលបំផុតក្នុងការបង្ហាញអត្ថប្រយោជន៍ ព្រោះវាអាចវាស់បានមុន និងក្រោយការផ្លាស់ប្តូរដោយឧបករណ៍វាស់ដូចគ្នា។

## របៀបគណនា

```
រយៈពេលដោះស្រាយ = ត្រាពេលវេលា(ការសម្រេចចាត់ថ្នាក់) − ត្រាពេលវេលា(ការដាក់ស្នើការបញ្ជូនបន្ត)

រាយការណ៍មធ្យមភាគ (median) និងភាគរយខ្ពស់ (ជាទូទៅ ភាគរយទី 90) មិនមែនតែមធ្យម (mean)
ព្រោះការចែកចាយមានទម្រង់ទាញទៅខាងស្តាំយ៉ាងខ្លាំង ដោយការបញ្ជូនបន្តដែលត្រូវបានត្រឡប់ ឬស្មុគស្មាញ។

ពិចារណាពេលវេលាតាមដំណាក់កាលរង នៅកន្លែងដែលប្រព័ន្ធកត់ត្រា៖
  ការដាក់ស្នើ → សេវាទទួលបាន
  ទទួលបាន → ការសម្រេចចាត់ថ្នាក់
  ការសម្រេចចាត់ថ្នាក់ → ការណាត់ជួបដែលបានកក់ (នៅកន្លែងពាក់ព័ន្ធ)
```

## ឧទាហរណ៍អនុវត្ត

ផ្លូវសវនកម្មនៃប្រព័ន្ធបញ្ជូនបន្តអេឡិចត្រូនិកមួយបង្ហាញថា រយៈពេលមធ្យមភាគពីការដាក់ស្នើដល់ការសម្រេចចាត់ថ្នាក់គឺ ១.៨ ថ្ងៃឆ្លងគ្រប់ឯកទេស ដោយមានរយៈពេលភាគរយទី 90 គឺ ៦ ថ្ងៃ ដែលបណ្តាលជាចម្បងមកពីការបញ្ជូនបន្តដែលត្រូវបានត្រឡប់ទៅអ្នកបញ្ជូនសម្រាប់ព័ត៌មានគ្លីនិកដែលខ្វះ។ ផ្លូវសុខភាពស្បែកពីចម្ងាយមួយដែលប្រើការចាត់ថ្នាក់ផ្អែកលើរូបភាពលើវេទិកាដូចគ្នា សម្រេចបានរយៈពេលដោះស្រាយមធ្យមភាគ ៤ ម៉ោង និងភាគរយទី 90 គឺ ១ ថ្ងៃ ព្រោះរូបថត និងប្រវត្តិមានរចនាសម្ព័ន្ធស្ទើរតែតែងគ្រប់គ្រាន់សម្រាប់ការសម្រេចចាត់ថ្នាក់ ដោយមិនត្រូវការការទាក់ទងបន្ថែម។

## ប្រភពទិន្នន័យ និងចំណុចត្រូវប្រុងប្រយ័ត្ន

ផ្លូវសវនកម្មរបស់ប្រព័ន្ធបញ្ជូនបន្តអេឡិចត្រូនិក ឬប្រព័ន្ធគ្រប់គ្រងការបញ្ជូនបន្តផ្ទាល់ គឺជាប្រភពចម្បង ដោយប្រើត្រាពេលវេលានៃការដាក់ស្នើ និងការសម្រេច។ អង្គការគួរបញ្ជាក់ថា "នាឡិកា" ផ្អាកខណៈពេលដែលការបញ្ជូនបន្តត្រូវបានត្រឡប់សម្រាប់ព័ត៌មានបន្ថែម ឬដើរជាប់លាប់ ព្រោះនិយមន័យទាំងពីរបង្កើតតួលេខខុសគ្នាគួរកត់សម្គាល់ សម្រាប់ដំណើរការមូលដ្ឋានដដែល។ រយៈពេលដោះស្រាយគួរត្រូវបានរាយការណ៍ជាពេលវេលាប្រតិទិន ឬម៉ោងធ្វើការជាប់លាប់ ព្រោះឥទ្ធិពលចុងសប្តាហ៍ និងថ្ងៃឈប់សម្រាកអាចបង្ខូចការប្រៀបធៀបរវាងសេវាដែលមានលំនាំការងារខុសគ្នា។

## កំហុសដែលតែងកើតមាន

- **វាស់តែការបញ្ជូនបន្ត "ស្អាត"**៖ ការដកការបញ្ជូនបន្តដែលត្រូវបានបដិសេធ ឬត្រឡប់ចេញពីការគណនា លាក់បន្ទុកកែតម្រូវ ដែលឧបករណ៍ឌីជីថលតែងត្រូវបានបម្រុងទុកដើម្បីកាត់បន្ថយ។
- **រាយការណ៍មធ្យម (mean) ជំនួសមធ្យមភាគ (median) និងភាគរយ**៖ ការបញ្ជូនបន្តដែលត្រូវបានត្រឡប់ និងដំណើរការយូរមួយចំនួនតូច នឹងទាញមធ្យមឱ្យខ្ពស់ឆ្ងាយពីបទពិសោធន៍ពិតរបស់អ្នកជំងឺធម្មតា។
- **ច្រឡំរយៈពេលដោះស្រាយជាមួយពេលរង់ចាំសរុប**៖ រយៈពេលដោះស្រាយគ្របដណ្តប់តែជំហានចាត់ថ្នាក់អាទិភាព បទពិសោធន៍សរុបរបស់អ្នកជំងឺក៏រួមបញ្ចូលបញ្ជីរង់ចាំគ្លីនិកខាងក្រោម ដែលជារង្វាស់ដាច់ដោយឡែក គ្រប់គ្រងដោយឧបសគ្គសមត្ថភាពដាច់ដោយឡែក។
- **មិនបែងចែកដំណាក់កាលរង**៖ សេវាដែលវាស់តែពេលវេលាចុងដល់ចុង មិនអាចប្រាប់ថាតើតួលេខយឺតបណ្តាលមកពីអ្នកបញ្ជូនដាក់ព័ត៌មានមិនពេញលេញ សមត្ថភាពចាត់ថ្នាក់របស់សេវាទទួល ឬទាំងពីរ។
