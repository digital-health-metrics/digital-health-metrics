# ក្របខ័ណ្ឌវាយតម្លៃសុខភាពឌីជីថលរបស់ WHO

អង្គការសុខភាពពិភពលោក (World Health Organization ឬ WHO) បោះពុម្ពសំណុំការណែនាំសម្រាប់វាយតម្លៃអន្តរាគមន៍សុខភាពឌីជីថលមុន អំឡុង និងក្រោយការដាក់ពង្រាយកម្រិតជាតិ ដែលគ្របដណ្តប់វិធីសាស្ត្រតាមដាន និងវាយតម្លៃ ការចាត់ថ្នាក់ស្តង់ដារនៃប្រភេទអន្តរាគមន៍ និងឧបករណ៍សម្រាប់វិនិច្ឆ័យថាតើកម្មវិធីសុខភាពឌីជីថលត្រៀមខ្លួនពង្រីកឬអត់ — ដោយសង្កត់ធ្ងន់ជាពិសេស និងជាប់លាប់លើសមធម៌ លទ្ធភាព និងសុវត្ថិភាព ដែលឆ្លុះបញ្ចាំងអាណត្តិរបស់ WHO ក្នុងការការពារសុខភាពប្រជាជនឆ្លងប្រទេសដែលមានហេដ្ឋារចនាសម្ព័ន្ធឌីជីថល និងប្រព័ន្ធសុខភាពខុសគ្នាខ្លាំង។ ខុសពីបញ្ជីត្រួតពិនិត្យដែលមានឈ្មោះតែមួយ នេះត្រូវបានយល់ល្អជាងជាគ្រួសារស៊ីសង្វាក់នៃឯកសារផ្សព្វផ្សាយរបស់ WHO ដែលអង្គការវាយតម្លៃការដាក់ពង្រាយសុខភាពឌីជីថលថ្នាក់ជាតិ ឬតំបន់ជាធម្មតាយកមកប្រើរួមគ្នា។

## ហេតុអ្វីបានជាសំខាន់

អន្តរាគមន៍សុខភាពឌីជីថលដែលដំណើរការល្អក្នុងបរិបទសាកល្បងដែលមានធនធានល្អ អាចបរាជ័យ ឬបង្កគ្រោះថ្នាក់យ៉ាងសកម្មនៅពេលដាក់ពង្រាយទូទាំងប្រទេសឆ្លងប្រជាជនដែលមានកម្រិតការតភ្ជាប់ ចំណេះដឹងឌីជីថល ភាសា និងសមត្ថភាពប្រព័ន្ធសុខភាពខុសគ្នាគួរកត់សម្គាល់ — ដែលជាគម្លាតដែលការណែនាំរបស់ WHO មានដើម្បីបិទ ព្រោះអាណត្តិស្នូលរបស់ WHO គឺសមធម៌សុខភាពប្រជាជនឆ្លងរដ្ឋសមាជិកដែលមានធនធានមិនស្មើគ្នាខ្លាំង មិនមែនផលប្រយោជន៍របស់ប្រព័ន្ធសុខភាព ឬអ្នកផ្គត់ផ្គង់ណាមួយឡើយ។ ការកំណត់ក្របខ័ណ្ឌរបស់ WHO ក៏ចាត់ទុកសុវត្ថិភាព និងលទ្ធភាពជាការពិចារណាត្រួតពិនិត្យទ្វារ មិនមែនគំនិតក្រោយដែលត្រូវវាយតម្លៃនៅពេលអន្តរាគមន៍ដំណើរការរួចហើយទេ៖ ឧបករណ៍សុខភាពឌីជីថលត្រូវតែបង្ហាញថាមិនបង្កើនវិសមភាព ឬបង្កើតហានិភ័យសុវត្ថិភាពថ្មីមុនពេលពិចារណាពង្រីក មិនមែនគ្រាន់តែបង្ហាញថាមានប្រសិទ្ធភាពរួមទេ។ សម្រាប់អង្គការណាដែលកំពុងធ្វើផែនការដាក់ពង្រាយសុខភាពឌីជីថលថ្នាក់ជាតិ ឬច្រើនប្រទេស ការតម្រឹមការវាយតម្លៃទៅនឹងការណែនាំរបស់ WHO ក៏តែងជាភាពចាំបាច់ជាក់ស្តែងផងដែរ ព្រោះវាជាក្របខ័ណ្ឌដែលទំនងត្រូវបានយោង ឬតម្រូវដោយក្រសួងសុខាភិបាល និងអ្នកផ្តល់មូលនិធិពហុភាគី។

## ឧទាហរណ៍អនុវត្ត

ក្រសួងសុខាភិបាលមួយគ្រោងពង្រីកសេវារំលឹកសារ SMS សុខភាពមាតាពីការសាកល្បងក្នុងតំបន់តែមួយទៅការដាក់ពង្រាយថ្នាក់ជាតិ។ ដោយអនុវត្តការណែនាំការចាត់ថ្នាក់របស់ WHO អន្តរាគមន៍ត្រូវបានចាត់ថ្នាក់ឱ្យច្បាស់ជាអន្តរាគមន៍សុខភាពឌីជីថលដែលប្រឈមមុខអតិថិជន ការទំនាក់ទំនងគោលដៅ ដែលអនុញ្ញាតឱ្យប្រៀបធៀបជាមួយអន្តរាគមន៍ស្រដៀងគ្នាដែលបានវាយតម្លៃកន្លែងផ្សេង ជាជាងត្រូវបានវាយតម្លៃដូចជាថ្មីទាំងស្រុង។ ផែនការតាមដាន និងវាយតម្លៃត្រូវបានសាងសង់តាមការណែនាំរបស់ WHO ដើម្បីតាមដានទាំងការទៅដល់ និងលទ្ធផលគ្លីនិក មិនមែនលទ្ធផលតែម្យ៉ាង។ ការវាយតម្លៃភាពត្រៀមខ្លួនពង្រីកកំណត់ថាភាពអាចទុកចិត្តបាននៃបណ្តាញទូរគមនាគមន៍ក្នុងតំបន់សាកល្បងខ្ពស់ជាងមធ្យមជាតិយ៉ាងខ្លាំង មានន័យថាការពិនិត្យសមធម៌ និងសុវត្ថិភាពដាក់ទង់ហានិភ័យសំខាន់៖ ការពង្រីកថ្នាក់ជាតិដោយមិនវាយតម្លៃការគ្របដណ្តប់បណ្តាញជនបទជាមុន អាចប្រមូលផ្តុំអត្ថប្រយោជន៍ក្នុងតំបន់ដែលតភ្ជាប់ល្អជាង និងមានបន្ទុកជាងរួចហើយ ដែលពង្រីកជាជាងបិទគម្លាតសមធម៌សុខភាពមាតាដែលមានស្រាប់។ ការរកឃើញនេះ — ដែលលេចចេញជាពិសេសព្រោះសមធម៌ត្រូវបានចាត់ទុកជាលក្ខណៈវិនិច្ឆ័យទ្វារ ជាជាងគំនិតក្រោយ — នាំឱ្យក្រសួងដាក់ការដាក់ពង្រាយជាដំណាក់កាលតាមកម្រិតការគ្របដណ្តប់បណ្តាញ ជាជាងពង្រីកស្មើគ្នា។

## ប្រភពទិន្នន័យ និងចំណុចត្រូវប្រុងប្រយ័ត្ន

ឯកសារណែនាំរបស់ WHO ត្រូវបានបោះពុម្ពជាសាធារណៈ និងអាចរកបានដោយឥតគិតថ្លៃ ប៉ុន្តែការអនុវត្តវាតម្រូវឱ្យមានទិន្នន័យក្នុងស្រុកដែល WHO ខ្លួនឯងមិនផ្គត់ផ្គង់ — ទិន្នន័យការតភ្ជាប់ និងចំណេះដឹងឌីជីថលជាក់លាក់សម្រាប់ប្រទេស ឬតំបន់ដាក់ពង្រាយ ទិន្នន័យសមត្ថភាពប្រព័ន្ធសុខភាពក្នុងស្រុក និងទិន្នន័យលទ្ធផលក្នុងស្រុក — មានន័យថាក្របខ័ណ្ឌផ្តល់រចនាសម្ព័ន្ធសម្រាប់ការវាយតម្លៃ មិនមែនទិន្នន័យខ្លួនឯងទេ។ ដោយសារការណែនាំរបស់ WHO ត្រូវបានធ្វើឱ្យទូទៅគ្រប់គ្រាន់ដោយចេតនាដើម្បីអនុវត្តឆ្លងបរិបទជាតិខុសគ្នាខ្លាំង ការអនុវត្តវាបានល្អតម្រូវឱ្យមានជំនាញ និងទិន្នន័យក្នុងស្រុកពិតប្រាកដ ដើម្បីជៀសវាងការវាយតម្លៃបែបទូទៅ បែបគូសធីក ដែលខកខានហានិភ័យជាក់លាក់ប្រទេស៖ ការវាយតម្លៃតម្រឹមតាម WHO ដែលអនុវត្តដោយគ្មានទិន្នន័យប្រព័ន្ធសុខភាព និងសមធម៌ក្នុងស្រុក ទំនងមិនលាតត្រដាងហានិភ័យដែលវាត្រូវបានរចនាដើម្បីចាប់ឡើយ។

## កំហុសដែលតែងកើតមាន

- **ចាត់ទុកការចាត់ថ្នាក់តែឯងជាការវាយតម្លៃ**៖ ការចាត់ថ្នាក់ប្រភេទអន្តរាគមន៍ឱ្យត្រឹមត្រូវគឺជាចំណុចចាប់ផ្តើមសម្រាប់ការប្រៀបធៀប មិនមែនជាការវាយតម្លៃសុវត្ថិភាព ប្រសិទ្ធភាព ឬភាពត្រៀមខ្លួនពង្រីករបស់អន្តរាគមន៍ដោយខ្លួនវាឡើយ។
- **ចាត់ទុកការវាយតម្លៃសមធម៌ជាការបន្ថែមចុងក្រោយ ជាជាងទ្វារត្រួតពិនិត្យ**៖ ការកំណត់ក្របខ័ណ្ឌរបស់ WHO ចាត់ទុកហានិភ័យសមធម៌ជាមូលហេតុដើម្បីបន្ថយល្បឿន ឬរចនាការដាក់ពង្រាយឡើងវិញ មិនមែនជាកំណត់សម្គាល់ត្រូវរាយការណ៍បន្ទាប់ពីការពង្រីកបានកើតឡើងរួច ការវាយតម្លៃវាតែបែបថយក្រោយបំផ្លាញគោលបំណងរបស់វា។
- **អនុវត្តក្របខ័ណ្ឌដោយគ្មានទិន្នន័យបរិបទក្នុងស្រុក**៖ ការណែនាំរបស់ WHO ទូទៅដោយចេតនា បើគ្មានទិន្នន័យការតភ្ជាប់ ចំណេះដឹង និងសមត្ថភាពប្រព័ន្ធសុខភាពក្នុងស្រុក ការវាយតម្លៃធៀបនឹងវាក្លាយជាលំហាត់បញ្ជីត្រួតពិនិត្យទូទៅ ជាជាងការវាយតម្លៃហានិភ័យពិតប្រាកដ។
- **សន្មតថាការតម្រឹមតាម WHO ជំនួសការត្រួតពិនិត្យនិយតកម្ម ឬក្រមសីលធម៌ថ្នាក់ជាតិ**៖ ការណែនាំរបស់ WHO ជូនដំណឹង និងរៀបចំរចនាសម្ព័ន្ធការវាយតម្លៃ ប៉ុន្តែមិនជំនួសដំណើរការអនុម័តនិយតកម្ម ការការពារទិន្នន័យ ឬក្រមសីលធម៌ដែលត្រូវការរបស់ប្រទេសខ្លួនឯងទេ។
